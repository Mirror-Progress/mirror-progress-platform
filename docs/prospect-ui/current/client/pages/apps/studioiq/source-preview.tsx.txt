import type { GetServerSideProps } from 'next';
import { readFile } from 'node:fs/promises';
import path from 'node:path';
import { createHash } from 'node:crypto';
import { getSessionAccountFromContext } from '../../../lib/auth-guards';
import { cognitoIsRequired } from '../../../lib/state-kernel/authorization/cognito';
import { findStudioIqProductAccessForAccount } from '../../../lib/studioiq-product-auth';

const previewPath = '/prospect/source-preview';

export default function StudioIqSourcePreviewPage() {
  // Authorized requests receive the complete standalone HTML document below.
  return null;
}

export const getServerSideProps: GetServerSideProps = async (context) => {
  const isLive = ['/prospect/sources', '/apps/studioiq/sources'].includes(context.resolvedUrl?.split('?')[0] || '');
  const nextQuery = new URLSearchParams();
  for (const key of ['opportunity', 'evidence', 'section', 'language']) {
    const value = context.query?.[key];
    if (typeof value === 'string') nextQuery.set(key, value);
  }
  const destination = isLive ? `/prospect/sources${nextQuery.size ? `?${nextQuery}` : ''}` : previewPath;
  context.res.setHeader('Cache-Control', 'private, no-store');
  context.res.setHeader('X-Robots-Tag', 'noindex, nofollow');
  const flag = `${process.env.STUDIOIQ_APP_ENABLED ?? ''}`.trim().toLowerCase();
  if (flag ? !['1', 'true', 'yes', 'on'].includes(flag) : process.env.NODE_ENV === 'production') {
    return { notFound: true };
  }

  const account = await getSessionAccountFromContext(context);
  if (!account) {
    return {
      redirect: {
        destination: cognitoIsRequired()
          ? `/api/auth/cognito/start?next=${encodeURIComponent(destination)}`
          : `/?auth=login&next=${encodeURIComponent(destination)}`,
        permanent: false,
      },
    };
  }
  const access = await findStudioIqProductAccessForAccount(account).catch(() => null);
  if (!access?.mirrorClientId) return { notFound: true };

  // This sample-only bundle has its own React and stylesheet versions. Serving
  // its HTML document avoids mixing them into the production application's root.
  // The entry document is not placed in public/ and is read only after access checks.
  const document = await readFile(path.join(process.cwd(), 'lib/studioiq-source-preview/index.html'), 'utf8');
  const workspaceKey = createHash('sha256').update(`${account.id}\0${access.mirrorClientId}`).digest('hex');
  const organization = (account.company || 'your firm').replace(/[&<>"']/g, (character) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[character]!));
  const html = document.replace('id="root"', `id="root" data-workspace-key="${workspaceKey}" data-organization-name="${organization}"`)
    .replace('<title>Prospect | Source intelligence preview</title>', `<title>Prospect | Source intelligence${isLive ? '' : ' preview'}</title>`);
  context.res.setHeader('Content-Type', 'text/html; charset=utf-8');
  context.res.statusCode = 200;
  context.res.end(html);
  return { props: {} };
};
