import type { NextApiRequest, NextApiResponse } from 'next';
import { pagesCookieResponse } from '@mirror-progress/identity-bridge';
import { identityBridge } from '../../../lib/state-kernel/authorization/identity';
import { identityEnabled } from '../../../lib/state-kernel/authorization/identity-provider';
import { prospectReturnTarget, rememberProspectReturn } from '../../../lib/state-kernel/authorization/prospect-return';
export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'GET') return res.status(405).end();
  res.setHeader('Cache-Control', 'no-store');
  if (process.env.SITE_SURFACE === 'campaign' || !identityEnabled()) return res.status(404).end();
  try {
    const target = prospectReturnTarget(req.query.next);
    const result = await identityBridge().beginAuthorization({ method: 'GET', returnTo: target.path }, pagesCookieResponse(res));
    rememberProspectReturn(res, result.authorizationUrl, target.full);
    return res.redirect(302, result.authorizationUrl);
  } catch { return res.redirect(302, 'https://accounts.mirrorprogress.com/?logout=1'); }
}
