import crypto from 'crypto';
import type { NextApiRequest, NextApiResponse } from 'next';
import { provisionStudioIqFounderPilot } from '../../../../lib/studioiq-founder-pilots';
import { identityEnabled } from '../../../../lib/state-kernel/authorization/identity-provider';

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  res.setHeader('Cache-Control', 'no-store');
  if (req.method !== 'POST' || !identityEnabled() || process.env.SITE_SURFACE === 'campaign') return res.status(404).end();
  const expected = process.env.IDENTITY_SESSION_STATUS_SECRET ?? '';
  const presented = `${req.headers.authorization ?? ''}`;
  if (expected.length < 48 || presented.length !== expected.length + 7 ||
      !crypto.timingSafeEqual(Buffer.from(presented), Buffer.from(`Bearer ${expected}`))) return res.status(404).end();
  const body = req.body ?? {};
  if (Object.keys(body).some((key) => !['name', 'email', 'company', 'trialDays', 'seatAllowance', 'actorPrincipalId', 'idempotencyKey'].includes(key))) {
    return res.status(400).json({ error: 'Invalid trial request.' });
  }
  const trialDays = Number(body.trialDays);
  const seatAllowance = Number(body.seatAllowance);
  if (!Number.isInteger(trialDays) || trialDays < 1 || trialDays > 365 ||
      !Number.isInteger(seatAllowance) || seatAllowance < 1 || seatAllowance > 10_000 ||
      typeof body.actorPrincipalId !== 'string' || !/^[a-f0-9-]{36}$/.test(body.actorPrincipalId) ||
      typeof body.idempotencyKey !== 'string' || !/^[a-f0-9-]{36}$/.test(body.idempotencyKey)) {
    return res.status(400).json({ error: 'Choose 1–365 trial days and at least one seat.' });
  }
  try {
    const pilot = await provisionStudioIqFounderPilot({
      idempotencyKey: body.idempotencyKey, organizationName: body.company, ownerEmail: body.email,
      ownerDisplayName: body.name, seatAllowance, trialDays, graceDays: 0,
      invitationMessage: `You have been invited to a ${trialDays}-day Prospect trial. Your trial begins when you first sign in.`,
      actorUserId: `identity:${body.actorPrincipalId}`,
      freshTrial: true,
    });
    if (pilot.trialDays !== trialDays || pilot.seatAllowance !== seatAllowance) {
      return res.status(409).json({ error: 'A trial for this recipient already exists with different terms.' });
    }
    return res.status(201).json({ pilotId: pilot.id, status: pilot.status });
  } catch (error) {
    return res.status(400).json({ error: error instanceof Error ? error.message : 'Unable to prepare trial.' });
  }
}
