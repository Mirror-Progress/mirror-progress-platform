import type { NextApiRequest, NextApiResponse } from 'next';
import { getAuthenticatedAccountFromRequest } from '../../../../lib/auth-guards';
import { getDb } from '../../../../lib/mongodb';
import { findStudioIqProductAccessForAccount } from '../../../../lib/studioiq-product-auth';
import { redactSearchText, type ProspectUsageEvent } from '../../../../lib/prospect-usage';

const validEvents = new Set<ProspectUsageEvent>(['screen_attention', 'discovery_search', 'project_open', 'project_feedback', 'agent_question', 'web_search', 'source_open']);
const validScreens = new Set(['discovery', 'source', 'globe', 'list', 'overview', 'practice', 'sources', 'agent']);
let ttlIndexReady: Promise<string> | undefined;
function sameOrigin(req: NextApiRequest) {
  if (req.headers['sec-fetch-site'] === 'cross-site') return false;
  if (!req.headers.origin) return true;
  try { return new URL(req.headers.origin).host === req.headers.host; } catch { return false; }
}

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  res.setHeader('Cache-Control', 'no-store');
  if (req.method !== 'POST') return res.status(405).json({ error: 'Method Not Allowed' });
  if (!req.headers['content-type']?.startsWith('application/json') || !sameOrigin(req)) return res.status(403).json({ error: 'Invalid request origin.' });
  const account = await getAuthenticatedAccountFromRequest(req);
  if (!account) return res.status(401).json({ error: 'Sign in to Prospect.' });
  const access = await findStudioIqProductAccessForAccount(account).catch(() => null);
  if (!access?.mirrorClientId) return res.status(403).json({ error: 'Prospect access required.' });
  if (req.headers.dnt === '1') return res.status(202).json({ ok: true });
  const event = req.body?.event as ProspectUsageEvent;
  if (!validEvents.has(event)) return res.status(400).json({ error: 'Invalid usage event.' });
  const raw = req.body?.detail;
  if (!raw || typeof raw !== 'object' || Array.isArray(raw) || JSON.stringify(raw).length > 1000) return res.status(400).json({ error: 'Invalid usage detail.' });
  const detail = {
    screen: typeof raw.screen === 'string' && validScreens.has(raw.screen) ? raw.screen : undefined,
    query: event === 'discovery_search' && typeof raw.query === 'string' ? redactSearchText(raw.query) : undefined,
    projectId: typeof raw.projectId === 'string' && /^opportunity_[a-zA-Z0-9-]{1,80}$/.test(raw.projectId) ? raw.projectId : undefined,
    feedback: event === 'project_feedback' && ['favorite_added', 'favorite_removed'].includes(raw.feedback) ? raw.feedback : undefined,
    durationSeconds: event === 'screen_attention' && Number.isFinite(raw.durationSeconds) ? Math.max(1, Math.min(300, Math.round(raw.durationSeconds))) : undefined,
  };
  const collection = (await getDb()).collection('studioiq_usage_events');
  ttlIndexReady ??= collection.createIndex({ expiresAt: 1 }, { expireAfterSeconds: 0 }).catch((error) => {
    ttlIndexReady = undefined;
    throw error;
  });
  await ttlIndexReady;
  const now = new Date();
  const start = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate()));
  if (await collection.countDocuments({ accountId: account.id, occurredAt: { $gte: start } }) >= 250) return res.status(429).json({ error: 'Usage event limit reached.' });
  await collection.insertOne({ accountId: account.id, organizationId: access.mirrorClientId, event, detail,
    occurredAt: now, expiresAt: new Date(Date.now() + 90 * 86400_000) });
  return res.status(202).json({ ok: true });
}
