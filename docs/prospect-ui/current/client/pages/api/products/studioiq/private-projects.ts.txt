import type { NextApiRequest, NextApiResponse } from 'next';
import { requireStudioIqProductAccess } from '../../../../lib/studioiq-product-auth';
import { deletePrivateProject, listPrivateProjects } from '../../../../lib/prospect-private-projects';

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (!['GET', 'DELETE'].includes(req.method || '')) return res.status(405).json({ error: 'Method Not Allowed' });
  const access = await requireStudioIqProductAccess(req, res);
  if (!access?.mirrorAccountId) return;
  try {
    if (req.method === 'GET') return res.status(200).json({ projects: await listPrivateProjects(access.mirrorAccountId) });
    const id = typeof req.body?.id === 'string' ? req.body.id : '';
    return res.status(200).json({ deleted: await deletePrivateProject(access.mirrorAccountId, id) });
  } catch { return res.status(503).json({ error: 'Private projects are temporarily unavailable.' }); }
}
