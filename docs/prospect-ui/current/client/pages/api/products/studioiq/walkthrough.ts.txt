import type { NextApiRequest, NextApiResponse } from 'next';
import { getAuthenticatedAccountFromRequest } from '../../../../lib/auth-guards';
import { getDb } from '../../../../lib/mongodb';
import { findStudioIqProductAccessForAccount } from '../../../../lib/studioiq-product-auth';
import { activeProspectWorkflow } from '../../../../lib/studioiq-walkthrough';

type Progress = { _id: string; version: number; status: 'completed' | 'skipped'; updatedAt: string };

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  res.setHeader('Cache-Control', 'no-store');
  if (!['GET', 'POST'].includes(req.method ?? '')) return res.status(405).json({ error: 'Method Not Allowed' });
  const account = await getAuthenticatedAccountFromRequest(req);
  if (!account) return res.status(401).json({ error: 'Sign in to Prospect.' });
  const access = await findStudioIqProductAccessForAccount(account).catch(() => null);
  if (!access) return res.status(403).json({ error: 'Prospect access required.' });
  const collection = (await getDb()).collection<Progress>('studioiq_walkthrough_progress');
  const id = `prospect:${account.id}:${activeProspectWorkflow.id}:v${activeProspectWorkflow.version}`;
  if (req.method === 'GET') {
    const progress = await collection.findOne({ _id: id });
    return res.status(200).json({ workflow: activeProspectWorkflow.id, version: activeProspectWorkflow.version,
      active: activeProspectWorkflow.status === 'active', status: progress?.status ?? 'new' });
  }
  if (Object.keys(req.body ?? {}).length !== 1 || !['completed', 'skipped'].includes(req.body?.status)) {
    return res.status(400).json({ error: 'Invalid walkthrough status.' });
  }
  await collection.updateOne({ _id: id }, { $set: {
    version: activeProspectWorkflow.version,
    status: req.body.status as Progress['status'],
    updatedAt: new Date().toISOString(),
  } }, { upsert: true });
  return res.status(200).json({ workflow: activeProspectWorkflow.id, version: activeProspectWorkflow.version, status: req.body.status });
}
