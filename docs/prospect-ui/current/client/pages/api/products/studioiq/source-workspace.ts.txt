import type { NextApiRequest, NextApiResponse } from 'next';
import { requireStudioIqProductAccess } from '../../../../lib/studioiq-product-auth';
import { loadProspectSourceWorkspace, validProspectProjectId } from '../../../../lib/prospect-source-workspace';

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  res.setHeader('Cache-Control', 'private, no-store');
  if (req.method !== 'GET') return res.status(405).json({ error: 'Method Not Allowed' });
  const access = await requireStudioIqProductAccess(req, res, { readOnlyAction: true });
  if (!access?.mirrorClientId) return;
  const id = req.query.opportunity;
  if (!validProspectProjectId(id)) return res.status(400).json({ error: 'Invalid project.' });
  try {
    const workspace = await loadProspectSourceWorkspace(access, id);
    if (!workspace) return res.status(404).json({ error: 'Project not found in your Prospect workspace.' });
    return res.status(200).json(workspace);
  } catch {
    return res.status(503).json({ error: 'This source workspace is temporarily unavailable.' });
  }
}
