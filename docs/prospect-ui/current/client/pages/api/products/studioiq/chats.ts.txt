import type { NextApiRequest, NextApiResponse } from 'next';
import { getAuthenticatedAccountFromRequest } from '../../../../lib/auth-guards';
import { getDb } from '../../../../lib/mongodb';
import { findStudioIqProductAccessForAccount } from '../../../../lib/studioiq-product-auth';
import { extendsChat } from '../../../../lib/prospect-chat-history';

type ChatMessage = { role: 'user' | 'assistant'; text: string };
type SourceContext = { opportunityId: string; evidenceId: string };
type ChatSession = { _id: string; accountId: string; conversationId: string; title: string; messages: ChatMessage[]; sourceContext?: SourceContext | null; createdAt: string; updatedAt: string };
const validId = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;
const validSourceContext = (value: unknown): value is SourceContext => Boolean(value && typeof value === 'object' && 'opportunityId' in value && 'evidenceId' in value &&
  typeof value.opportunityId === 'string' && /^opportunity_[a-zA-Z0-9-]+$/.test(value.opportunityId) &&
  typeof value.evidenceId === 'string' && /^evidence_[a-zA-Z0-9-]+$/.test(value.evidenceId));

export const config = { api: { bodyParser: { sizeLimit: '10mb' } } };

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  res.setHeader('Cache-Control', 'no-store');
  if (!['GET', 'PUT'].includes(req.method ?? '')) return res.status(405).json({ error: 'Method Not Allowed' });
  const account = await getAuthenticatedAccountFromRequest(req);
  if (!account) return res.status(401).json({ error: 'Sign in to Prospect.' });
  const access = await findStudioIqProductAccessForAccount(account).catch(() => null);
  if (!access) return res.status(403).json({ error: 'Prospect access required.' });
  const collection = (await getDb()).collection<ChatSession>('studioiq_chat_sessions');

  if (req.method === 'GET') {
    const id = typeof req.query.id === 'string' ? req.query.id : undefined;
    if (id) {
      if (!validId.test(id)) return res.status(400).json({ error: 'Invalid chat ID.' });
      const chat = await collection.findOne({ _id: `${account.id}:${id}`, accountId: account.id });
      if (!chat) return res.status(404).json({ error: 'Chat not found.' });
      return res.status(200).json({ id: chat.conversationId, title: chat.title, messages: chat.messages, sourceContext: chat.sourceContext, updatedAt: chat.updatedAt });
    }
    const chats = await collection.find({ accountId: account.id }, { projection: { conversationId: 1, title: 1, updatedAt: 1 } })
      .sort({ updatedAt: -1 }).toArray();
    return res.status(200).json({ chats: chats.map((chat) => ({ id: chat.conversationId, title: chat.title, updatedAt: chat.updatedAt })) });
  }

  const { id, messages, sourceContext } = req.body ?? {};
  if (typeof id !== 'string' || !validId.test(id) || !Array.isArray(messages) || messages.length < 1 || messages.length > 2000 ||
    (sourceContext !== undefined && !validSourceContext(sourceContext)) ||
    messages.some((item: unknown) => !item || typeof item !== 'object' || !('role' in item) || !('text' in item) ||
      !['user', 'assistant'].includes(String(item.role)) || typeof item.text !== 'string' || item.text.length > 100000) ||
    JSON.stringify(messages).length > 9000000) return res.status(400).json({ error: 'Invalid chat.' });
  const entries = messages as ChatMessage[];
  const title = entries.find((entry) => entry.role === 'user')?.text.trim().slice(0, 80) || 'New chat';
  const updatedAt = new Date().toISOString();
  const key = `${account.id}:${id}`;
  const existing = await collection.findOne({ _id: key, accountId: account.id });
  if (existing) {
    if (!extendsChat(existing.messages, entries)) return res.status(409).json({ error: 'This chat changed elsewhere. Reload it before continuing.' });
    const result = await collection.updateOne({ _id: key, accountId: account.id, updatedAt: existing.updatedAt }, {
      $set: { title, messages: entries, sourceContext: sourceContext ?? null, updatedAt },
    });
    if (!result.matchedCount) return res.status(409).json({ error: 'This chat changed elsewhere. Reload it before continuing.' });
  } else {
    await collection.updateOne({ _id: key, accountId: account.id }, {
      $setOnInsert: { accountId: account.id, conversationId: id, title, messages: entries, sourceContext: sourceContext ?? null, createdAt: updatedAt, updatedAt },
    }, { upsert: true });
  }
  return res.status(200).json({ id, title, updatedAt });
}
