import Link from 'next/link';
import { useCallback, useEffect, useRef, useState, type CSSProperties, type KeyboardEvent, type PointerEvent as ReactPointerEvent } from 'react';
import {
  CheckIcon,
  ArrowTopRightOnSquareIcon,
  ChevronDownIcon,
  CreditCardIcon,
  GlobeAltIcon,
  MagnifyingGlassIcon,
  MoonIcon,
  QuestionMarkCircleIcon,
  SunIcon,
  UserCircleIcon,
  XMarkIcon,
} from '@heroicons/react/24/outline';
import { useTheme } from '../../hooks/useTheme';
import {
  StudioIqDiscovery,
  type DiscoveryOpportunity,
  type DiscoverySummary,
} from './StudioIqDiscovery';
import styles from './StudioIqTerminal.module.css';
import { ProspectAgentPanel } from './ProspectAgentPanel';
import type { MarketSnapshot, MarketSnapshotItem } from '../../lib/prospect-agent/market-snapshot';
import { activeProspectWorkflow, availableProspectTourSteps, type ProspectTourScene } from '../../lib/studioiq-walkthrough';
import { ProspectWalkthrough } from './ProspectWalkthrough';

type DiscoveryResponse = {
  generated_at_utc: string;
  total: number;
  can_favorite: boolean;
  summary: DiscoverySummary;
  opportunities: DiscoveryOpportunity[];
};

const featuredMarketCodes = ['WLD', 'USA', 'GBR', 'CAN', 'AUS', 'BRA', 'MEX'];

async function readJson<T>(response: Response): Promise<T> {
  const body = await response.json().catch(() => ({}));
  if (!response.ok) {
    throw new Error(
      body.error || `Prospect request failed (${response.status}).`
    );
  }
  return body as T;
}

export function StudioIqTerminal(props: {
  accountName: string;
  organizationName: string;
  canManageBilling: boolean;
  localPreview?: boolean;
  trialLocked?: boolean;
  initialDiscovery?: Pick<DiscoveryResponse, 'opportunities' | 'summary' | 'can_favorite'>;
}) {
  const { theme, toggleTheme } = useTheme();
  const [items, setItems] = useState<DiscoveryOpportunity[]>(props.initialDiscovery?.opportunities ?? []);
  const [summary, setSummary] = useState<DiscoverySummary | null>(props.initialDiscovery?.summary ?? null);
  const [canFavorite, setCanFavorite] = useState(props.initialDiscovery?.can_favorite ?? false);
  const [loading, setLoading] = useState(!props.initialDiscovery);
  const [agentOpen, setAgentOpen] = useState(true);
  const [agentExpanded, setAgentExpanded] = useState(false);
  const [pendingScene, setPendingScene] = useState<ProspectTourScene | null>(null);
  const [pendingSequence, setPendingSequence] = useState<ProspectTourScene[] | null>(null);
  const [tourSequence, setTourSequence] = useState<ProspectTourScene[] | null>(null);
  const [chatShare, setChatShare] = useState(0.34);
  const [layoutWidth, setLayoutWidth] = useState(1440);
  const layoutRef = useRef<HTMLDivElement>(null);
  const [error, setError] = useState<string | null>(null);
  const [marketCountry, setMarketCountry] = useState('WLD');
  const [marketSnapshot, setMarketSnapshot] = useState<MarketSnapshot | null>(null);
  const [marketDetail, setMarketDetail] = useState<MarketSnapshotItem | null>(null);
  const marketDetailCloseRef = useRef<HTMLButtonElement>(null);
  const marketDetailOriginRef = useRef<HTMLButtonElement | null>(null);
  const [marketError, setMarketError] = useState(false);
  const [marketPickerOpen, setMarketPickerOpen] = useState(false);
  const [marketQuery, setMarketQuery] = useState('');
  const marketPickerRef = useRef<HTMLDivElement>(null);
  const marketSearchRef = useRef<HTMLInputElement>(null);
  const marketListRef = useRef<HTMLDivElement>(null);
  const [accountMenuOpen, setAccountMenuOpen] = useState(false);
  const [tourIndex, setTourIndex] = useState<number | null>(null);
  const tourChecked = useRef(false);
  const tourSteps = availableProspectTourSteps({ hasProjects: items.length > 0, canFavorite });
  const activeTourSteps = tourSequence ? tourSequence.map((scene) => tourSteps.find((step) => step.id === scene)).filter((step): step is (typeof tourSteps)[number] => Boolean(step)) : tourSteps;
  const tourScene = tourIndex === null ? null : activeTourSteps[tourIndex]?.id ?? null;
  const expandedLayout = agentOpen && agentExpanded && tourScene === null;
  useEffect(() => {
    if (loading || props.trialLocked || tourChecked.current || activeProspectWorkflow.status !== 'active') return;
    tourChecked.current = true;
    const forced = new URLSearchParams(window.location.search).get('tour') === '1';
    const key = `prospect-tour-${activeProspectWorkflow.id}-v${activeProspectWorkflow.version}`;
    if (props.localPreview) {
      if (forced || !window.localStorage.getItem(key)) setTourIndex(0);
      return;
    }
    let cancelled = false;
    void fetch('/api/products/studioiq/walkthrough', { cache: 'no-store' })
      .then((response) => response.ok ? response.json() : null)
      .then((progress) => { if (!cancelled && progress?.active && (forced || progress.status === 'new')) setTourIndex(0); })
      .catch(() => undefined);
    return () => { cancelled = true; };
  }, [loading, props.localPreview, props.trialLocked]);
  useEffect(() => {
    if (tourScene === 'agent') { setAgentOpen(true); setAgentExpanded(false); }
  }, [tourScene]);
  const finishTour = useCallback((status: 'completed' | 'skipped') => {
    setTourIndex(null);
    setTourSequence(null);
    if (tourSequence) { setAgentOpen(true); return; }
    if (props.localPreview) window.localStorage.setItem(`prospect-tour-${activeProspectWorkflow.id}-v${activeProspectWorkflow.version}`, status);
    else void fetch('/api/products/studioiq/walkthrough', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ status }) });
  }, [props.localPreview, tourSequence]);
  const nextTourStep = useCallback(() => {
    if (tourIndex === null) return;
    if (tourIndex >= activeTourSteps.length - 1) finishTour('completed');
    else setTourIndex(tourIndex + 1);
  }, [activeTourSteps.length, finishTour, tourIndex]);
  const startTourScene = useCallback((scene: ProspectTourScene) => {
    if (activeProspectWorkflow.status !== 'active') return;
    const index = tourSteps.findIndex((step) => step.id === scene);
    if (index < 0) return;
    setAgentExpanded(false);
    setTourSequence(null);
    setTourIndex(index);
  }, [tourSteps]);
  const showTourScene = useCallback((scene: ProspectTourScene) => {
    if (!agentOpen) { setPendingScene(scene); return; }
    if (!tourSteps.some((step) => step.id === scene)) return;
    setAgentExpanded(false);
    setTourSequence([scene]);
    setTourIndex(0);
  }, [agentOpen, tourSteps]);
  const showTourSequence = useCallback((scenes: ProspectTourScene[]) => {
    if (activeProspectWorkflow.status !== 'active' || scenes.length < 2 || scenes.some((scene) => !tourSteps.some((step) => step.id === scene))) return;
    if (!agentOpen) { setPendingSequence(scenes); return; }
    setAgentExpanded(false);
    setTourSequence(scenes);
    setTourIndex(0);
  }, [agentOpen, tourSteps]);
  const closeAgent = useCallback(() => {
    setAgentOpen(false);
    setAgentExpanded(false);
    if (tourScene === 'agent') finishTour('skipped');
  }, [finishTour, tourScene]);
  useEffect(() => {
    if (!agentOpen) return;
    if (pendingSequence) { setPendingSequence(null); showTourSequence(pendingSequence); return; }
    if (pendingScene) { setPendingScene(null); showTourScene(pendingScene); }
  }, [agentOpen, pendingScene, pendingSequence, showTourScene, showTourSequence]);
  const accountMenuRef = useRef<HTMLDivElement>(null);
  const accountInitial =
    props.accountName.trim().charAt(0).toUpperCase() || 'A';
  const visibleMarket = marketSnapshot?.geography.code === marketCountry ? marketSnapshot : null;
  const marketCountries = marketSnapshot?.countries.length ? marketSnapshot.countries : [{ code: 'WLD', name: 'World' }];
  const selectedMarketName = marketCountries.find((country) => country.code === marketCountry)?.name ?? marketCountry;
  const filteredMarkets = marketCountries.filter((country) =>
    `${country.name} ${country.code}`.toLowerCase().includes(marketQuery.trim().toLowerCase()));
  const featuredMarkets = filteredMarkets.filter((country) => featuredMarketCodes.includes(country.code));
  const otherMarkets = filteredMarkets.filter((country) => !featuredMarketCodes.includes(country.code));

  useEffect(() => {
    if (!marketDetail) return;
    marketDetailCloseRef.current?.focus();
    const closeOnEscape = (event: globalThis.KeyboardEvent) => {
      if (event.key === 'Escape') closeMarketDetail();
    };
    document.addEventListener('keydown', closeOnEscape);
    return () => document.removeEventListener('keydown', closeOnEscape);
  }, [marketDetail]);

  function closeMarketDetail() {
    setMarketDetail(null);
    requestAnimationFrame(() => marketDetailOriginRef.current?.focus());
  }

  useEffect(() => {
    if (!marketPickerOpen) return;
    marketSearchRef.current?.focus();
    marketListRef.current?.scrollTo({ top: 0 });
    const closePicker = (event: MouseEvent) => {
      if (!marketPickerRef.current?.contains(event.target as Node)) setMarketPickerOpen(false);
    };
    const closeOnEscape = (event: globalThis.KeyboardEvent) => {
      if (event.key === 'Escape') setMarketPickerOpen(false);
    };
    document.addEventListener('pointerdown', closePicker);
    document.addEventListener('keydown', closeOnEscape);
    return () => { document.removeEventListener('pointerdown', closePicker); document.removeEventListener('keydown', closeOnEscape); };
  }, [marketPickerOpen]);

  useEffect(() => { marketListRef.current?.scrollTo({ top: 0 }); }, [marketQuery]);

  useEffect(() => {
    let active = true;
    const loadMarket = async () => {
      try {
        const endpoint = props.localPreview ? '/api/local/prospect-market' : '/api/products/studioiq/market';
        const response = await fetch(`${endpoint}?country=${encodeURIComponent(marketCountry)}`, { cache: 'no-store' });
        const result = await readJson<MarketSnapshot>(response);
        if (active) { setMarketSnapshot(result); setMarketError(false); }
      } catch {
        if (active) setMarketError(true);
      }
    };
    void loadMarket();
    const timer = window.setInterval(() => void loadMarket(), 60 * 60 * 1000);
    return () => { active = false; window.clearInterval(timer); };
  }, [marketCountry, props.localPreview]);

  const loadDiscovery = useCallback(async () => {
    if (props.initialDiscovery) {
      setItems(props.initialDiscovery.opportunities);
      setSummary(props.initialDiscovery.summary);
      setCanFavorite(props.initialDiscovery.can_favorite);
      setLoading(false);
      return;
    }
    setLoading(true);
    setError(null);
    try {
      const result = await readJson<DiscoveryResponse>(
        await fetch('/api/products/studioiq/discovery')
      );
      setItems(result.opportunities);
      setSummary(result.summary);
      setCanFavorite(result.can_favorite);
    } catch (reason) {
      setError(
        reason instanceof Error
          ? reason.message
          : 'Unable to load Prospect discovery.'
      );
    } finally {
      setLoading(false);
    }
  }, [props.initialDiscovery]);

  useEffect(() => {
    void loadDiscovery();
  }, [loadDiscovery]);

  useEffect(() => {
    if (window.matchMedia('(max-width: 767px)').matches) setAgentOpen(false);
  }, []);

  useEffect(() => {
    const saved = Number(window.localStorage.getItem('prospect-chat-share-v1'));
    if (saved >= 0.2 && saved <= 0.65) setChatShare(saved);
    const observer = new ResizeObserver(([entry]) => setLayoutWidth(entry.contentRect.width));
    if (layoutRef.current) observer.observe(layoutRef.current);
    return () => observer.disconnect();
  }, []);

  const setChatShareSaved = (share: number) => {
    const next = Math.max(0.2, Math.min(0.65, share));
    setChatShare(next);
    window.localStorage.setItem('prospect-chat-share-v1', String(next));
  };
  const resizeChat = (event: ReactPointerEvent<HTMLDivElement>) => {
    const handle = event.currentTarget;
    handle.setPointerCapture(event.pointerId);
    const onMove = (move: PointerEvent) => {
      const rect = layoutRef.current?.getBoundingClientRect();
      if (!rect) return;
      setChatShareSaved((rect.right - move.clientX) / rect.width);
    };
    const onEnd = () => {
      handle.removeEventListener('pointermove', onMove);
      handle.removeEventListener('pointerup', onEnd);
      handle.removeEventListener('pointercancel', onEnd);
    };
    handle.addEventListener('pointermove', onMove);
    handle.addEventListener('pointerup', onEnd);
    handle.addEventListener('pointercancel', onEnd);
  };
  const onChatResizeKey = (event: KeyboardEvent<HTMLDivElement>) => {
    if (event.key === 'ArrowLeft' || event.key === 'ArrowRight') {
      event.preventDefault();
      setChatShareSaved(chatShare + (event.key === 'ArrowLeft' ? 0.025 : -0.025));
    } else if (event.key === 'Home' || event.key === 'End') {
      event.preventDefault();
      setChatShareSaved(event.key === 'Home' ? 0.2 : 0.65);
    }
  };
  const chatWidth = Math.max(320, Math.min(700, layoutWidth * chatShare, layoutWidth - 650));

  useEffect(() => {
    const closeAccountMenu = (event: PointerEvent) => {
      if (!accountMenuRef.current?.contains(event.target as Node))
        setAccountMenuOpen(false);
    };
    const closeAccountMenuWithKeyboard = (event: globalThis.KeyboardEvent) => {
      if (event.key === 'Escape') setAccountMenuOpen(false);
    };
    document.addEventListener('pointerdown', closeAccountMenu);
    document.addEventListener('keydown', closeAccountMenuWithKeyboard);
    return () => {
      document.removeEventListener('pointerdown', closeAccountMenu);
      document.removeEventListener('keydown', closeAccountMenuWithKeyboard);
    };
  }, []);

  return (
    <div className={styles.shell} style={{ '--prospect-top-offset': '102px' } as CSSProperties}>
      <header className={styles.header}>
        <Link
          className={styles.brand}
          href="/apps/studioiq"
          aria-label="Prospect discovery home"
        >
          <span className={styles.brandMark} aria-hidden="true" />
          <div>
            <strong>Prospect</strong>
            <small className={styles.brandByline}>by <span className={styles.brandWatermark} role="img" aria-label="Mirror Progress" /></small>
          </div>
        </Link>
        <div className={styles.headerActions}>
          <button className={styles.agentLauncher} type="button" data-active={agentOpen && tourScene === null} onClick={() => { if (tourScene) { finishTour('skipped'); setAgentOpen(true); setAgentExpanded(false); } else if (agentOpen) setAgentExpanded((current) => !current); else { setAgentOpen(true); setAgentExpanded(false); } }} aria-label="Open Prospect assistant chat"><span>Ask</span><span className={styles.agentMark} role="img" aria-label="MP" /></button>
          <div className={styles.accountMenu} ref={accountMenuRef}>
            <button
              className={styles.menuToggle}
              type="button"
              aria-label="Open Prospect menu"
              aria-haspopup="menu"
              aria-expanded={accountMenuOpen}
              onClick={() => setAccountMenuOpen((current) => !current)}
            >
              <span className={styles.menuGlyph} aria-hidden="true"><span /><span /></span>
            </button>
            {accountMenuOpen ? (
              <div
                className={styles.accountPopover}
                role="menu"
                aria-label="Prospect navigation"
              >
                <div>
                  <span>Signed in</span>
                  <strong>{props.accountName}</strong>
                  <small>{props.organizationName}</small>
                </div>
                <Link href="/apps/studioiq" role="menuitem">
                  <GlobeAltIcon />
                  Discovery
                </Link>
                <Link href="/apps/studioiq/account" role="menuitem">
                  <UserCircleIcon />
                  Account information
                </Link>
                <button type="button" role="menuitem" onClick={() => { setAccountMenuOpen(false); startTourScene('intro'); }}>
                  <QuestionMarkCircleIcon />
                  Replay walkthrough
                </button>
                {props.canManageBilling ? (
                  <Link href="/apps/studioiq/billing" role="menuitem">
                    <CreditCardIcon />
                    Billing
                  </Link>
                ) : null}
              </div>
            ) : null}
          </div>
          <Link
            className={styles.identity}
            href="/apps/studioiq/account"
            aria-label={`Open ${props.accountName} account`}
            title="Account information"
          >
            <span className={styles.identityInitial} aria-hidden="true">{accountInitial}</span>
            <span className={styles.identityStatus} aria-hidden="true" />
          </Link>
          <button
            className={styles.themeToggle}
            type="button"
            onClick={toggleTheme}
            aria-label={`Switch to ${theme === 'dark' ? 'light' : 'dark'} theme`}
            title={`${theme === 'dark' ? 'Light' : 'Dark'} mode`}
          >
            <span className={styles.themeGlyph} aria-hidden="true"><SunIcon className={styles.sunIcon} /><MoonIcon className={styles.moonIcon} /></span>
          </button>
        </div>
      </header>

      <div className={styles.marketTicker} aria-label="Sourced market context with observation periods; figures are not project-specific">
          <span className={styles.marketTickerLabel}><span aria-hidden="true" />MARKET PULSE</span>
          <div className={styles.marketTickerViewport}>
            <div className={styles.marketTickerTrack}>
              {[0, 1].map((copy) => (
                <div className={styles.marketTickerGroup} key={copy} aria-hidden={copy === 1}>
                  {(visibleMarket?.items.length ? visibleMarket.items : []).map((item) => (
                    <button className={styles.marketTickerItem} type="button" key={item.id} onClick={(event) => { marketDetailOriginRef.current = event.currentTarget; setMarketDetail(item); }} tabIndex={copy === 1 ? -1 : undefined} aria-label={`Details for ${item.label}, ${item.geography}, ${item.period}`} title={`${item.label}: ${item.value}. ${item.geography}, ${item.period}. Open explanation and source.`}>
                      <span className={styles.marketTickerName}>{item.label}</span>
                      <strong className={styles.marketTickerValue}>{item.value}</strong>
                      <span className={styles.marketTickerRegion}>{item.geography} ·</span>
                      <span className={styles.marketTickerPeriod}>{item.period}</span>
                    </button>
                  ))}
                  {!visibleMarket?.items.length && copy === 0 ? <span className={styles.marketTickerItem}>{marketError || visibleMarket?.mode === 'unavailable' ? 'Market sources unavailable' : 'Loading market sources…'}</span> : null}
                  {copy === 0 && visibleMarket && marketCountry !== 'WLD' && !visibleMarket.items.some((item) => item.id.includes(marketCountry)) ? <span className={styles.marketTickerItem}>No current country series for {visibleMarket.geography.name}</span> : null}
                </div>
              ))}
            </div>
          </div>
          {marketDetail ? <div className={styles.marketDetailBackdrop} onMouseDown={(event) => { if (event.target === event.currentTarget) closeMarketDetail(); }}>
            <section className={styles.marketDetail} role="dialog" aria-modal="true" aria-labelledby="market-detail-heading" aria-describedby="market-detail-explanation">
              <div className={styles.marketDetailTop}><span>MARKET PULSE / SOURCE CONTEXT</span><button ref={marketDetailCloseRef} type="button" onClick={closeMarketDetail} aria-label="Close market detail"><XMarkIcon /></button></div>
              <div className={styles.marketDetailHero}><div><p>{marketDetail.geography} · {marketDetail.frequency === 'annual' ? 'Annual series' : 'Monthly series'}</p><h2 id="market-detail-heading">{marketDetail.label}</h2></div><strong>{marketDetail.value}</strong></div>
              <dl><div><dt>Observation</dt><dd>{marketDetail.period}</dd></div><div><dt>Published by</dt><dd>{marketDetail.sourceLabel}</dd></div><div><dt>Retrieved</dt><dd>{marketDetail.fetchedAt.slice(0, 10)}</dd></div></dl>
              <p id="market-detail-explanation" className={styles.marketDetailExplanation}>{marketDetail.interpretation}</p>
              <p className={styles.marketDetailScope}>This is market context for {marketDetail.geography}, not a measurement of any individual Prospect project.</p>
              <a href={marketDetail.sourceUrl} target="_blank" rel="noopener noreferrer">Open original dataset <ArrowTopRightOnSquareIcon /></a>
            </section>
          </div> : null}
          <div className={styles.marketTickerStatus}>
            {visibleMarket?.stale || marketError ? <span className={styles.marketTickerStale}>SOURCE DELAY</span> : null}
            <div className={styles.marketPicker} ref={marketPickerRef}>
              <button type="button" className={styles.marketPickerTrigger} aria-label={`Market pulse: ${selectedMarketName}. Choose market`} aria-expanded={marketPickerOpen} aria-controls="prospect-market-options" onClick={() => { setMarketPickerOpen((open) => !open); setMarketQuery(''); }}>
                <GlobeAltIcon aria-hidden="true" />
                <span>{selectedMarketName}</span>
                <ChevronDownIcon aria-hidden="true" />
              </button>
              {marketPickerOpen ? <div className={styles.marketPickerPopover} id="prospect-market-options" aria-label="Choose market">
                <div className={styles.marketPickerSearch}>
                  <MagnifyingGlassIcon aria-hidden="true" />
                  <input ref={marketSearchRef} type="search" aria-label="Search countries" placeholder="Search markets" value={marketQuery} onChange={(event) => setMarketQuery(event.target.value)} />
                </div>
                <div className={styles.marketPickerList} ref={marketListRef}>
                  {featuredMarkets.length ? <div className={styles.marketPickerGroup}><span className={styles.marketPickerGroupLabel}>FEATURED</span>{featuredMarkets.map((country) => <button key={country.code} type="button" className={styles.marketPickerOption} data-selected={country.code === marketCountry} onClick={() => { setMarketCountry(country.code); setMarketPickerOpen(false); }}><span>{country.name}</span><small>{country.code}</small>{country.code === marketCountry ? <CheckIcon aria-hidden="true" /> : null}</button>)}</div> : null}
                  {otherMarkets.length ? <div className={styles.marketPickerGroup}><span className={styles.marketPickerGroupLabel}>ALL MARKETS</span>{otherMarkets.map((country) => <button key={country.code} type="button" className={styles.marketPickerOption} data-selected={country.code === marketCountry} onClick={() => { setMarketCountry(country.code); setMarketPickerOpen(false); }}><span>{country.name}</span><small>{country.code}</small>{country.code === marketCountry ? <CheckIcon aria-hidden="true" /> : null}</button>)}</div> : null}
                  {!filteredMarkets.length ? <p className={styles.marketPickerEmpty}>No matching market</p> : null}
                </div>
                <p className={styles.marketPickerFoot}>Country figures use their latest available year.</p>
              </div> : null}
            </div>
          </div>
      </div>

      {error ? (
        <div className={styles.error} role="alert">
          <span>{error}</span>
          <button type="button" onClick={() => void loadDiscovery()}>
            Retry
          </button>
        </div>
      ) : null}

      <div ref={layoutRef} className={`${styles.workspaceLayout} ${expandedLayout ? styles.workspaceLayoutExpanded : ''}`} style={{ '--prospect-chat-width': `${chatWidth}px` } as CSSProperties}>
        <div className={styles.workspaceMain}><StudioIqDiscovery
          items={items}
          summary={summary}
          canFavorite={canFavorite}
          loading={loading}
          theme={theme}
          localPreview={props.localPreview}
          onReload={loadDiscovery}
          tourScene={tourScene}
        /></div>
        {expandedLayout ? <div className={styles.workspaceResize} role="separator" aria-label="Resize Prospect and chat" aria-orientation="vertical" aria-valuenow={Math.round(chatShare * 100)} aria-valuemin={20} aria-valuemax={65} tabIndex={0} onPointerDown={resizeChat} onKeyDown={onChatResizeKey} onDoubleClick={() => setChatShareSaved(0.34)} /> : null}
        <aside className={`${styles.agentDock} ${expandedLayout ? styles.agentDockExpanded : ''} ${!agentOpen || (tourScene && tourScene !== 'agent') ? styles.agentDockTourHidden : ''}`} data-tour-anchor={agentOpen ? 'agent' : undefined} aria-label="Prospect assistant chat" aria-hidden={!agentOpen || Boolean(tourScene && tourScene !== 'agent')}><ProspectAgentPanel endpoint={props.localPreview ? '/api/local/prospect-agent' : '/api/products/studioiq/agent'} localLab={Boolean(props.localPreview)} expanded={expandedLayout} onToggleExpand={() => setAgentExpanded((current) => !current)} onClose={closeAgent} availableScenes={tourSteps.map((step) => step.id)} onShowScene={showTourScene} onShowSequence={showTourSequence} /></aside>
      </div>
      {tourIndex !== null && activeTourSteps[tourIndex] ? <ProspectWalkthrough key={activeTourSteps[tourIndex].id} step={activeTourSteps[tourIndex]} index={tourIndex} total={activeTourSteps.length} onNext={nextTourStep} onBack={() => setTourIndex((current) => current === null ? null : Math.max(0, current - 1))} onSkip={() => finishTour('skipped')} onUnavailable={nextTourStep} /> : null}
    </div>
  );
}
