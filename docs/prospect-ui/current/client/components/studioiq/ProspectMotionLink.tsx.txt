import Link, { type LinkProps } from 'next/link';
import { useRouter } from 'next/router';
import type { AnchorHTMLAttributes, MouseEvent } from 'react';

type Props = LinkProps & Omit<AnchorHTMLAttributes<HTMLAnchorElement>, keyof LinkProps>;

/** Keep Discovery and its Source workspace in one visual navigation sequence. */
export function ProspectMotionLink({ href, onClick, children, ...props }: Props) {
  const router = useRouter();

  function navigate(event: MouseEvent<HTMLAnchorElement>) {
    onClick?.(event);
    if (event.defaultPrevented || event.button !== 0 || event.metaKey || event.ctrlKey || event.shiftKey || event.altKey || props.target === '_blank') return;
    const start = document.startViewTransition;
    if (!start || window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    event.preventDefault();
    const destination = typeof href === 'string' ? href : router.pathname;
    if (destination === router.asPath) return;
    document.documentElement.dataset.prospectTransition = 'source';
    const transition = start.call(document, () => router.push(href).then(() => undefined));
    void transition.finished.then(
      () => { delete document.documentElement.dataset.prospectTransition; },
      () => { delete document.documentElement.dataset.prospectTransition; }
    );
  }

  return <Link href={href} onClick={navigate} {...props}>{children}</Link>;
}
