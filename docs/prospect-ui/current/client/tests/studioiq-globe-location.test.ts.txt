import test from 'node:test';
import assert from 'node:assert/strict';
import { globeProjectLocation } from '../lib/studioiq-globe-location';
import { clusterGlobeProjects, type GlobeOpportunity } from '../lib/studioiq-globe-context';

test('missing city coordinates resolve to an explicit country marker without inventing a city', () => {
  const location = globeProjectLocation({ country: 'Poland', location: null });
  assert.equal(location?.coordinate_method, 'country_centroid');
  assert.ok(location && location.latitude > 49 && location.latitude < 55);
  assert.equal(globeProjectLocation({ country: null, location: null }), null);
  assert.equal(globeProjectLocation({ country: 'Unknown', location: null }), null);
  assert.ok(globeProjectLocation({ country: 'USA', location: null }));
});
test('recorded city coordinates remain untouched', () => {
  const location = { latitude: 52.52, longitude: 13.405, coordinate_method: 'city_centroid', coordinate_confidence: 'approximate' };
  assert.equal(globeProjectLocation({ country: 'Germany', location }), location);
});
test('unresolved cities share their country marker and never merge with a city pin', () => {
  const location = globeProjectLocation({ country: 'Poland', location: null })!;
  const project = (id: string, city: string, method: string): GlobeOpportunity => ({
    id, city, country: 'Poland', title: id, latitude: location.latitude, longitude: location.longitude,
    coordinateMethod: method, active: false, value: 0, projectType: null,
  });
  const groups = clusterGlobeProjects([project('a', 'City A', 'country_centroid'), project('b', 'City B', 'country_centroid'), project('c', 'City C', 'city_centroid')]);
  assert.deepEqual(groups.map((group) => group.items.length), [2, 1]);
});

test('recorded small cities resolve within the right country and preserve source coordinates', () => {
  const meersburg = globeProjectLocation({city: 'Meersburg', country: 'Germany', location: null})!;
  assert.equal(meersburg.coordinate_method, 'city_centroid');
  assert.ok(Math.abs(meersburg.latitude - 47.6942) < 0.001);
  const menton = globeProjectLocation({city: 'Menton', country: 'France', location: null})!;
  assert.ok(Math.abs(menton.longitude - 7.5044) < 0.001);
  assert.equal(globeProjectLocation({city:'Menton',country:'Germany',location:null})?.coordinate_method,'country_centroid');
  const source = {latitude:47.7,longitude:9.3,coordinate_method:'source_document',coordinate_confidence:'high'};
  assert.equal(globeProjectLocation({city:'Meersburg',country:'Germany',location:source}), source);
});
test('qualified names, accents, ambiguity and overseas locations retain their geography', () => {
  const locate = (city:string,country:string,title?:string) => globeProjectLocation({city,country,title,location:null})!;
  assert.equal(locate('Malmo','Sweden').coordinate_method,'city_centroid');
  assert.ok(locate('Richmond, British Columbia','Canada').longitude < -120);
  assert.equal(locate('Richmond, Ontario','Canada').coordinate_method,'country_centroid');
  assert.equal(locate('Birkenfeld','Germany').coordinate_method,'country_centroid');
  assert.ok(Math.abs(locate('Birkenfeld','Germany','Birkenfeld outdoor pool comprehensive renovation').latitude - 49.6525) < .001);
  assert.ok(locate('Les Abymes','France').longitude < -60);
  assert.equal(locate('British Columbia','Canada').coordinate_method,'region_centroid');
  assert.ok(locate('Riviera / Veveyse','Switzerland').longitude < 7);
});
test('missing city fields use only reviewed unambiguous names, never arbitrary title fragments', () => {
  assert.equal(globeProjectLocation({country:'Canada',title:'Quayside Toronto Public Library branch',location:null})?.coordinate_method,'city_centroid');
  assert.equal(globeProjectLocation({country:'Mexico',title:'Nationwide programme',location:null})?.coordinate_method,'country_centroid');
  assert.equal(globeProjectLocation({country:'Mexico',title:'Toronto architecture office',location:null})?.coordinate_method,'country_centroid');
});
test('city aliases sharing coordinates stay in one completely selectable group', () => {
  const p = (id:string,city:string): GlobeOpportunity => ({id,city,country:'Canada',title:id,latitude:45.5088,longitude:-73.5878,coordinateMethod:'city_centroid',active:false,value:0,projectType:null});
  const groups = clusterGlobeProjects([p('one','Montreal'),p('two','Montréal, Quebec')],3.6);
  assert.equal(groups.length,1);
  assert.deepEqual(groups[0].items.map(p=>p.id),['one','two']);
});

test('production California projects use city and regional references instead of one country marker', () => {
  const places = [
    'Larkspur, California', 'Los Angeles, California', 'West Sacramento, California',
    'Pomona, California', 'Pasadena, California', 'Pasadena, California',
    'Rancho Cordova, California', 'Stockton, California', 'Greenville, California',
    'Arcadia, California', 'Dinuba, California', 'Merced, California',
    'Los Alamitos, California', 'Hollister, California', 'Sunnyvale, California',
    'San Francisco, California', 'Redwood City, California', 'Riverside, California',
    'Sunnyvale, California', 'Orange County, California', 'Moffett Field, California',
    'Porterville / Tule River Reservation, California', 'Long Beach, California',
    'Glendora, California', 'San Francisco, California', 'San Diego, California',
    'Sacramento / South Pacific Division, California', 'Corona, California',
    'Fresno', 'Marina, California', 'San Francisco, California', 'Fresno',
  ];
  const items: GlobeOpportunity[] = places.map((city, index) => {
    const location = globeProjectLocation({city,country:'United States',title:city,location:null});
    assert.ok(location, city);
    assert.notEqual(location.coordinate_method, 'country_centroid', city);
    return {id:String(index),title:city,city,country:'United States',...location,
      coordinateMethod:location.coordinate_method,active:false,value:0,projectType:null};
  });
  const groups = clusterGlobeProjects(items,3.6);
  assert.equal(groups.reduce((total,group)=>total+group.items.length,0),32);
  assert.ok(groups.length>10, `expected a dispersed map, got ${groups.length} groups`);
  assert.equal(new Set(groups.flatMap(group=>group.items.map(item=>item.id))).size,32);
  const sf = items.find(item=>item.city==='San Francisco, California')!;
  const sacramento = items.find(item=>item.city==='Sacramento / South Pacific Division, California')!;
  assert.ok(sf.longitude < -122 && sf.latitude > 37);
  assert.ok(sacramento.latitude > 38 && sacramento.longitude > -122);
  assert.equal(sacramento.coordinateMethod,'region_centroid');
});

test('reviewed city inventory resolves at production density without losing projects', async () => {
  const {default: anchors} = await import('../data/studioiq-city-anchors.json');
  const items = Array.from({length:130}, (_, i): GlobeOpportunity => {
    const a = anchors[i % anchors.length];
    const location = globeProjectLocation({city:a.city,country:a.country,title:a.titleHint ?? '',location:null})!;
    assert.ok(location && Number.isFinite(location.latitude) && Number.isFinite(location.longitude));
    assert.equal(location.coordinate_method,a.kind === 'region' ? 'region_centroid' : 'city_centroid',a.city);
    return {id:String(i),title:a.city,city:a.city,country:a.country,...location,coordinateMethod:location.coordinate_method,active:false,value:0,projectType:null};
  });
  for (const zoom of [1,1.55,2.25,3.1,3.6]) {
    const groups = clusterGlobeProjects(items,zoom);
    assert.equal(groups.reduce((n,g)=>n+g.items.length,0),130);
    assert.equal(new Set(groups.flatMap(g=>g.items.map(p=>p.id))).size,130);
  }
  const germany = items.filter(p=>p.country === 'Germany');
  assert.ok(clusterGlobeProjects(germany,3.6).length >= 15);
});
