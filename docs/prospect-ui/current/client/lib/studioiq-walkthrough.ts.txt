export type ProspectWorkflow = {
  id: string;
  version: number;
  status: 'active' | 'retired';
  steps: readonly ProspectTourStep[];
};

export const activeProspectWorkflow: ProspectWorkflow = {
  id: PROSPECT_WALKTHROUGH_ID,
  version: PROSPECT_WALKTHROUGH_VERSION,
  status: 'active',
  steps: prospectTourSteps,
};

export function availableProspectTourSteps(input: { hasProjects: boolean; canFavorite: boolean }) {
  if (activeProspectWorkflow.status !== 'active') return [];
  return activeProspectWorkflow.steps.filter((step) =>
    (!step.requiresProject || input.hasProjects) && (!step.requiresFavoriteAccess || input.canFavorite));
}

export function resolveActiveProspectScene(requested: unknown, availableScenes: readonly ProspectTourScene[]): ProspectTourScene | null {
  if (activeProspectWorkflow.status !== 'active' || typeof requested !== 'string') return null;
  if (!activeProspectWorkflow.steps.some((step) => step.id === requested)) return null;
  return availableScenes.includes(requested as ProspectTourScene) ? requested as ProspectTourScene : null;
}

// Screen actions are driven by the active workflow, never by arbitrary DOM selectors
// supplied by the model. Retired or unavailable scenes cannot enter a guided path.
export function resolveActiveProspectSequence(requested: unknown, availableScenes: readonly ProspectTourScene[]): ProspectTourScene[] | null {
  if (activeProspectWorkflow.status !== 'active' || !Array.isArray(requested) || requested.length < 2 || requested.length > 8) return null;
  const scenes = requested.map((scene) => resolveActiveProspectScene(scene, availableScenes));
  if (scenes.some((scene) => scene === null) || new Set(scenes).size !== scenes.length) return null;
  return scenes as ProspectTourScene[];
}
