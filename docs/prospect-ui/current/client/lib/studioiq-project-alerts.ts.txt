import crypto from 'crypto';
import { SendEmailCommand, SESv2Client } from '@aws-sdk/client-sesv2';
import { findAccountById } from './accounts';
import { getDb } from './mongodb';
import { listPublishedStudioIqIntelligence } from './research-intelligence';
import { qualifiesForProjectAlert } from './studioiq-project-alert-policy';
import { authorizationCollectionNames } from './state-kernel/authorization/store';

type Preference = {
  _id: string;
  mirrorAccountId: string;
  mirrorClientId: string;
  email: string;
  enabled: boolean;
  enabledAt: string | null;
  updatedAt: string;
};

type Delivery = {
  _id: string;
  mirrorAccountId: string;
  candidateId: string;
  status: 'pending' | 'sending' | 'sent' | 'failed';
  attempts: number;
  updatedAt: string;
  sentAt?: string;
  messageId?: string;
  error?: string;
};

const SENDER = 'hello@mirrorprogress.com';
const sender = new SESv2Client({});

function escapeHtml(value: string) {
  return value.replace(/[&<>"']/g, (character) => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;',
  })[character] ?? character);
}

export async function projectAlertPreference(accountId: string) {
  const db = await getDb();
  return db.collection<Preference>('studioiq_project_alert_preferences').findOne({ _id: accountId });
}

export async function updateProjectAlertPreference(input: {
  accountId: string; mirrorClientId: string; email: string; enabled: boolean;
}) {
  const db = await getDb();
  const collection = db.collection<Preference>('studioiq_project_alert_preferences');
  const current = await collection.findOne({ _id: input.accountId });
  const now = new Date().toISOString();
  const enabledAt = input.enabled ? (current?.enabled ? current.enabledAt ?? now : now) : null;
  await collection.updateOne({ _id: input.accountId }, {
    $set: {
      mirrorAccountId: input.accountId,
      mirrorClientId: input.mirrorClientId,
      email: input.email,
      enabled: input.enabled,
      enabledAt,
      updatedAt: now,
    },
  }, { upsert: true });
  return { enabled: input.enabled, enabledAt };
}

function projectUrl(id: string) {
  const base = (process.env.PLATFORM_BASE_URL || 'https://platform.mirrorprogress.com').replace(/\/$/, '');
  return `${base}/prospect?opportunity=${encodeURIComponent(id)}`;
}

async function sendProjectAlert(input: {
  to: string;
  candidate: Awaited<ReturnType<typeof listPublishedStudioIqIntelligence>>['opportunities'][number]['candidate'];
  brief: NonNullable<Awaited<ReturnType<typeof listPublishedStudioIqIntelligence>>['opportunities'][number]['pursuitBrief']>;
}) {
  const { candidate, brief } = input;
  const title = candidate.title;
  const location = [candidate.city, candidate.country].filter(Boolean).join(', ');
  const url = projectUrl(candidate.id);
  const settingsUrl = `${(process.env.PLATFORM_BASE_URL || 'https://platform.mirrorprogress.com').replace(/\/$/, '')}/prospect/account`;
  const reason = brief.strategicRationale[0] || brief.summary || candidate.valueReasons[0] || 'This project is a strong match for your practice.';
  const subject = `A strong Prospect match: ${title}`;
  const text = `A newly published project is a strong match for your practice.\n\n${title}${location ? `\n${location}` : ''}${candidate.deadline ? `\nDeadline: ${candidate.deadline}` : ''}\n\nWhy this matches: ${reason}\n\nView project: ${url}\n\nManage project alerts: ${settingsUrl}`;
  const html = `<div style="background:#0c1920;padding:32px 16px;font-family:Arial,sans-serif;color:#e9f1eb"><div style="max-width:560px;margin:auto;border:1px solid #344a45;border-radius:18px;background:#15252a;padding:32px"><p style="color:#b8dd9b;letter-spacing:.16em;font-size:12px;font-weight:700">MIRROR PROGRESS / PROSPECT</p><h1 style="font-size:28px;line-height:1.15;margin:18px 0">A project worth a closer look</h1><p style="color:#b9c9c4">A newly published project is a strong match for your practice.</p><div style="padding:20px;margin:24px 0;border:1px solid #3d5850;border-radius:12px;background:#1b3033"><h2 style="margin:0 0 8px;font-size:21px">${escapeHtml(title)}</h2>${location ? `<p style="color:#b9c9c4;margin:0 0 12px">${escapeHtml(location)}</p>` : ''}<p style="line-height:1.5;margin:0">${escapeHtml(reason)}</p>${candidate.deadline ? `<p style="color:#b9c9c4;margin:14px 0 0">Deadline: ${escapeHtml(candidate.deadline)}</p>` : ''}</div><a href="${escapeHtml(url)}" style="display:inline-block;background:#c9ef9e;color:#14201d;text-decoration:none;font-weight:700;padding:14px 20px;border-radius:9px">View in Prospect →</a><p style="font-size:12px;color:#93a8a2;margin:28px 0 0">You chose to receive strong project matches. <a style="color:#c9ef9e" href="${escapeHtml(settingsUrl)}">Manage alerts</a></p></div></div>`;
  return sender.send(new SendEmailCommand({
    FromEmailAddress: `Mirror Progress <${SENDER}>`,
    Destination: { ToAddresses: [input.to] },
    ReplyToAddresses: [SENDER],
    Content: { Simple: {
      Subject: { Data: subject, Charset: 'UTF-8' },
      Body: { Text: { Data: text, Charset: 'UTF-8' }, Html: { Data: html, Charset: 'UTF-8' } },
    } },
  }));
}

export async function runStudioIqProjectAlerts() {
  const db = await getDb();
  const preferences = await db.collection<Preference>('studioiq_project_alert_preferences')
    .find({ enabled: true, enabledAt: { $type: 'string' } }).toArray();
  const deliveries = db.collection<Delivery>('studioiq_project_alert_deliveries');
  await deliveries.updateMany({ status: 'sending', updatedAt: { $lt: new Date(Date.now() - 30 * 60_000).toISOString() } }, {
    $set: { status: 'failed', error: 'Recovered an interrupted delivery.', updatedAt: new Date().toISOString() },
  });
  const cache = new Map<string, Awaited<ReturnType<typeof listPublishedStudioIqIntelligence>>>();
  let sent = 0; let skipped = 0; let failed = 0;
  for (const preference of preferences) {
    if (!preference.enabledAt || !preference.email || !preference.mirrorClientId) continue;
    let sentForRecipient = 0;
    const [membership, access, legacyAccount, principal] = await Promise.all([
      db.collection<{ mirrorAccountId: string; mirrorClientId: string; status: string; emailLower: string }>('studioiq_workspace_memberships').findOne({ mirrorAccountId: preference.mirrorAccountId, mirrorClientId: preference.mirrorClientId, status: 'active' }),
      db.collection<{ mirrorAccountId: string; mirrorClientId: string; productId: string; status: string }>('product_access').findOne({ mirrorAccountId: preference.mirrorAccountId, mirrorClientId: preference.mirrorClientId, productId: 'studioiq', status: 'active' }),
      findAccountById(preference.mirrorAccountId),
      db.collection<{ principalId: string; status: string; email: string }>(authorizationCollectionNames.principals).findOne({ principalId: preference.mirrorAccountId }),
    ]);
    const identity = legacyAccount ?? principal;
    if (!membership || !access || identity?.status !== 'active' || identity.email.toLowerCase() !== preference.email.toLowerCase() || membership.emailLower !== preference.email.toLowerCase()) { skipped++; continue; }
    let catalog = cache.get(preference.mirrorClientId);
    if (!catalog) {
      catalog = await listPublishedStudioIqIntelligence({ mirrorClientId: preference.mirrorClientId });
      cache.set(preference.mirrorClientId, catalog);
    }
    for (const item of catalog.opportunities) {
      if (!qualifiesForProjectAlert(item.candidate, item.pursuitBrief, preference.enabledAt)) continue;
      if (sentForRecipient >= 3) break;
      const id = crypto.createHash('sha256').update(`${preference.mirrorAccountId}:${item.candidate.id}`).digest('hex');
      const now = new Date().toISOString();
      await deliveries.updateOne({ _id: id }, { $setOnInsert: {
        mirrorAccountId: preference.mirrorAccountId, candidateId: item.candidate.id,
        status: 'pending', attempts: 0, updatedAt: now,
      } }, { upsert: true });
      const claimed = await deliveries.findOneAndUpdate({ _id: id, status: { $in: ['pending', 'failed'] }, attempts: { $lt: 3 } }, {
        $set: { status: 'sending', updatedAt: now }, $inc: { attempts: 1 },
      }, { returnDocument: 'after' });
      if (!claimed) { skipped++; continue; }
      const freshPreference = await projectAlertPreference(preference.mirrorAccountId);
      if (!freshPreference?.enabled || freshPreference.enabledAt !== preference.enabledAt) {
        await deliveries.updateOne({ _id: id }, { $set: { status: 'failed', updatedAt: now } });
        skipped++;
        continue;
      }
      try {
        const response = await sendProjectAlert({ to: preference.email, candidate: item.candidate, brief: item.pursuitBrief! });
        await deliveries.updateOne({ _id: id }, { $set: { status: 'sent', sentAt: new Date().toISOString(), updatedAt: new Date().toISOString(), messageId: response.MessageId } });
        sent++;
        sentForRecipient++;
      } catch (error) {
        await deliveries.updateOne({ _id: id }, { $set: { status: 'failed', error: error instanceof Error ? error.message.slice(0, 200) : 'Email failed', updatedAt: new Date().toISOString() } });
        failed++;
      }
      if (sent + failed >= 50) return { examined: preferences.length, sent, skipped, failed };
    }
  }
  return { examined: preferences.length, sent, skipped, failed };
}
