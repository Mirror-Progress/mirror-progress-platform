export type ProspectUsageEvent = 'screen_attention' | 'discovery_search' | 'project_open' | 'project_feedback' | 'agent_question' | 'web_search' | 'source_open';

export function recordProspectUsage(event: ProspectUsageEvent, detail: Record<string, string | number | boolean> = {}) {
  if (typeof window === 'undefined' || window.location.pathname.startsWith('/local/')) return;
  if (navigator.doNotTrack === '1') return;
  void fetch('/api/products/studioiq/usage', {
    method: 'POST', headers: { 'content-type': 'application/json' }, credentials: 'same-origin', keepalive: true,
    body: JSON.stringify({ event, detail }),
  }).catch(() => undefined);
}

export function redactSearchText(value: string) {
  return value.trim().replace(/[\w.+-]+@[\w.-]+\.[A-Za-z]{2,}/g, '[email]')
    .replace(/\+?\d[\d().\s-]{7,}\d/g, '[phone]').slice(0, 160);
}
