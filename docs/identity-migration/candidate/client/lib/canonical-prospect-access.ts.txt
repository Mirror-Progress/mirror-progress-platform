import crypto from 'crypto';
import type { StoredAccount } from './accounts';
import { getDb, getMongoClient } from './mongodb';
import { ensureMongoBootstrap } from './mongodb-bootstrap';
import { findProductEntitlement, productAccessMode } from './product-commercial';
import { findProductAccess, type ProductAccessRecord } from './product-access';
import { identityEnabled } from './state-kernel/authorization/identity-provider';
import { AuthorizationStore } from './state-kernel/authorization/store';
import {
  MIRROR_PROGRESS_INTERNAL_TENANT_ID,
  MIRROR_PROGRESS_INTERNAL_WORKSPACE_ID,
} from './state-kernel/types';

const internalScope = {
  tenantId: MIRROR_PROGRESS_INTERNAL_TENANT_ID,
  workspaceId: MIRROR_PROGRESS_INTERNAL_WORKSPACE_ID,
};

/** Only active internal canonical members can join the existing internal Prospect workspace. */
export async function ensureCanonicalProspectAccess(account: StoredAccount): Promise<ProductAccessRecord | null> {
  if (!identityEnabled() || account.status !== 'active' || !['super_admin', 'admin', 'project_lead'].includes(account.role)) return null;
  const store = new AuthorizationStore();
  const [principal, membership] = await Promise.all([
    store.getPrincipal(account.id), store.getMembership(account.id, internalScope),
  ]);
  if (principal?.status !== 'active' || membership?.status !== 'active') return null;
  const role = membership.roles.includes('super_admin') ? 'owner'
    : membership.roles.includes('admin') ? 'admin'
      : membership.roles.includes('project_lead') ? 'member' : null;
  if (!role) return null;

  await ensureMongoBootstrap();
  const db = await getDb();
  const mapped = await db.collection('authorization_identity_mappings').findOne({
    principalId: account.id, issuer: process.env.IDENTITY_ISSUER, enabled: true,
  });
  if (!mapped || typeof mapped.reviewRef !== 'string' || !mapped.reviewRef) return null;
  const existing = await findProductAccess({ productId: 'studioiq', mirrorAccountId: account.id });
  if (existing) return existing; // Never revive disabled or invited access.

  const anchors = await db.collection<ProductAccessRecord>('product_access').find({
    productId: 'studioiq', status: 'active', 'metadata.bootstrap': 'production_owner',
  }).limit(2).toArray();
  if (anchors.length !== 1) throw new Error('Canonical Prospect workspace is unavailable.');
  const anchor = anchors[0]!;
  const anchorMembership = await store.getMembership(anchor.mirrorAccountId, internalScope);
  if (!anchor.mirrorClientId || !anchor.externalOrganizationId ||
      !(await db.collection('clients').findOne({ id: anchor.mirrorClientId })) ||
      anchorMembership?.status !== 'active' || !anchorMembership.roles.includes('super_admin')) {
    throw new Error('Canonical Prospect workspace is invalid.');
  }
  const entitlement = await findProductEntitlement({ productId: 'studioiq', mirrorClientId: anchor.mirrorClientId, refresh: false });
  if (!entitlement || productAccessMode(entitlement.status) === 'none') throw new Error('Canonical Prospect access is inactive.');

  const now = new Date().toISOString();
  const digest = crypto.createHash('sha256').update(`${account.id}:${anchor.mirrorClientId}`).digest('hex').slice(0, 24);
  const accessId = `studioiq_internal_access_${digest}`;
  const auditId = `studioiq_internal_access_audit_${digest}`;
  const access: ProductAccessRecord & { _id: string } = {
    _id: accessId, id: accessId, productId: 'studioiq', mirrorAccountId: account.id,
    mirrorClientId: anchor.mirrorClientId, externalUserId: `mirror:${account.id}`,
    externalOrganizationId: anchor.externalOrganizationId, status: 'active',
    metadata: { identityAuthority: 'mirror_progress', workspaceRole: role, bootstrap: 'canonical_internal' },
    createdAt: now, updatedAt: now,
  };
  const audit = {
    _id: auditId, id: auditId, actorUserId: account.id,
    action: 'studioiq_internal_access_from_canonical_membership',
    entityType: 'ProductAccess', entityId: accessId, sourceSystem: 'platform',
    metadata: { mirrorClientId: anchor.mirrorClientId, workspaceRole: role }, createdAt: now,
  };
  const session = (await getMongoClient()).startSession();
  try {
    await session.withTransaction(async () => {
      const collection = db.collection<typeof access>('product_access');
      if (await collection.findOne({ productId: 'studioiq', mirrorAccountId: account.id }, { session })) return;
      await collection.insertOne(access, { session });
      await db.collection<typeof audit>('audit_events').insertOne(audit, { session });
    });
  } catch (error) {
    if (!(error && typeof error === 'object' && 'code' in error && error.code === 11000)) throw error;
  } finally {
    await session.endSession();
  }
  return findProductAccess({ productId: 'studioiq', mirrorAccountId: account.id });
}
