import type { NextApiRequest, NextApiResponse } from 'next';
import { beginCognitoAuthorization } from '../../../../lib/state-kernel/authorization/cognito';

export default function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'GET') return res.status(405).json({ error: 'method_not_allowed' });
  if (`${process.env.SITE_SURFACE ?? ''}` === 'campaign') return res.status(404).end('Not Found');
  try {
    return res.redirect(302, beginCognitoAuthorization(req, res));
  } catch {
    return res.status(503).json({ error: 'authentication_unavailable' });
  }
}
